/** Вход для Android Auto в категории навигации: приборы рисуются на всей поверхности. */
class NavCarAppService : CarAppService() {
    override fun createHostValidator(): HostValidator = HostValidator.ALLOW_ALL_HOSTS_VALIDATOR

    override fun onCreateSession(): Session = NavSession()
}

class NavSession : Session() {
    private var conn: ServiceConnection? = null

    override fun onCreateScreen(intent: Intent): Screen {
        // запускаем и держим службу связи с адаптером, пока открыт экран
        ObdBridgeService.start(carContext)
        val c = object : ServiceConnection {
            override fun onServiceConnected(name: ComponentName?, service: IBinder?) {}
            override fun onServiceDisconnected(name: ComponentName?) {}
        }
        conn = c
        try {
            carContext.bindService(Intent(carContext, ObdBridgeService::class.java), c, Context.BIND_AUTO_CREATE)
        } catch (e: Exception) {
            conn = null
        }
        NavData.start()

        carContext.getCarService(AppManager::class.java).setSurfaceCallback(NavRenderer())

        lifecycle.addObserver(object : DefaultLifecycleObserver {
            override fun onDestroy(owner: LifecycleOwner) {
                NavData.stop()
                conn?.let {
                    try {
                        carContext.unbindService(it)
                    } catch (e: Exception) {
                    }
                }
                conn = null
            }
        })
        return NavScreen(carContext)
    }
}

class NavScreen(carContext: CarContext) : Screen(carContext) {
    override fun onGetTemplate(): Template {
        val exit = Action.Builder()
            .setTitle("Выход")
            .setOnClickListener { carContext.finishCarApp() }
            .build()
        return NavigationTemplate.Builder()
            .setActionStrip(ActionStrip.Builder().addAction(exit).build())
            .build()
    }
}

package com.aatorque.nav

import android.os.SystemClock
import com.aatorque.obd.ObdBridgeService
import java.util.concurrent.ConcurrentHashMap

/** Фоновый опрос значений из ObdBridgeService. Рисование читает только готовые значения и не ждёт адаптер. */
object NavData {
    private val FAST = arrayOf("0d", "0c", "05")
    private val SLOW = arrayOf("ff9010", "ff9003", "ff9005", "ff9011")

    private val values = ConcurrentHashMap<String, Double>()
    @Volatile private var running = false
    private var thread: Thread? = null

    fun get(id: String): Double? = values[id]

    @Synchronized
    fun start() {
        if (running) return
        running = true
        val t = Thread({ loop() }, "nav-data")
        t.isDaemon = true
        thread = t
        t.start()
    }

    @Synchronized
    fun stop() {
        running = false
        thread?.interrupt()
        thread = null
    }

    private fun put(id: String, v: Double?) {
        if (v == null || v.isNaN()) values.remove(id) else values[id] = v
    }

    private fun loop() {
        var lastSlow = 0L
        while (running) {
            try {
                for (id in FAST) put(id, ObdBridgeService.peek(id))
                val now = SystemClock.uptimeMillis()
                if (now - lastSlow > 1000L) {
                    for (id in SLOW) put(id, ObdBridgeService.peek(id))
                    lastSlow = now
                }
                Thread.sleep(30)
            } catch (e: InterruptedException) {
                return
            } catch (e: Exception) {
                try {
                    Thread.sleep(500)
                } catch (ie: InterruptedException) {
                    return
                }
            }
        }
    }
}

package com.aatorque.nav

import android.graphics.Canvas
import android.graphics.Color
import android.graphics.Paint
import android.graphics.Rect
import android.graphics.RectF
import android.os.Handler
import android.os.HandlerThread
import android.view.Choreographer
import android.view.Surface
import androidx.car.app.SurfaceCallback
import androidx.car.app.SurfaceContainer
import kotlin.math.PI
import kotlin.math.cos
import kotlin.math.exp
import kotlin.math.max
import kotlin.math.min
import kotlin.math.roundToInt
import kotlin.math.sin

/**
 * Приборы на реальных данных. Рисование идёт в своём потоке по сигналу вертикальной синхронизации,
 * показания сглажены, раскладка выбирается по форме стабильной области.
 */
class NavRenderer : SurfaceCallback {
    private class Gauge(
        val label: String,
        val unit: String,
        val id: String,
        val scale: Double,
        val min: Float,
        val max: Float
    ) {
        var shown = min
        var has = false
        var text = "--"
        var textKey = Int.MIN_VALUE
    }

    private class Info(val label: String, val id: String, val unit: String, val decimals: Int)

    private val gauges = listOf(
        Gauge("Скорость", "км/ч", "0d", 1.0, 0f, 240f),
        Gauge("Обороты", "x1000", "0c", 0.001, 0f, 6f),
        Gauge("Темп. ОЖ", "°C", "05", 1.0, 40f, 120f)
    )
    private val infos = listOf(
        Info("DPF", "ff9010", "%", 0),
        Info("Сажа", "ff9003", "г/л", 2),
        Info("АКПП", "ff9005", "°C", 0),
        Info("Интервал", "ff9011", "км", 0)
    )

    private val debugOverlay = false

    @Volatile private var surface: Surface? = null
    @Volatile private var width = 0
    @Volatile private var height = 0
    @Volatile private var dpi = 0
    @Volatile private var visible = Rect()
    @Volatile private var stable = Rect()
    @Volatile private var debug = ""

    private var thread: HandlerThread? = null
    private var handler: Handler? = null
    private var choreographer: Choreographer? = null
    @Volatile private var running = false
    private var lastNanos = 0L
    private var hardware = true

    private val rectF = RectF()
    private val fill = Paint().apply { style = Paint.Style.FILL; isAntiAlias = true }
    private val stroke = Paint().apply { style = Paint.Style.STROKE; isAntiAlias = true }
    private val text = Paint().apply { isAntiAlias = true; textAlign = Paint.Align.CENTER }

    private val frame = object : Choreographer.FrameCallback {
        override fun doFrame(frameTimeNanos: Long) {
            if (!running) return
            val dt = if (lastNanos == 0L) 0.016f
            else ((frameTimeNanos - lastNanos) / 1e9f).coerceIn(0.001f, 0.1f)
            lastNanos = frameTimeNanos
            draw(dt)
            choreographer?.postFrameCallback(this)
        }
    }

    override fun onSurfaceAvailable(surfaceContainer: SurfaceContainer) {
        surface = surfaceContainer.surface
        width = surfaceContainer.width
        height = surfaceContainer.height
        dpi = surfaceContainer.dpi
        updateDebug()
        start()
    }

    override fun onSurfaceDestroyed(surfaceContainer: SurfaceContainer) {
        stop()
        surface = null
    }

    override fun onVisibleAreaChanged(visibleArea: Rect) {
        visible = Rect(visibleArea)
        updateDebug()
    }

    override fun onStableAreaChanged(stableArea: Rect) {
        stable = Rect(stableArea)
        updateDebug()
    }

    private fun updateDebug() {
        debug = "${width}x$height $dpi dpi  visible ${visible.toShortString()}  stable ${stable.toShortString()}"
    }

    private fun start() {
        if (running) return
        running = true
        lastNanos = 0L
        val th = HandlerThread("nav-render")
        th.start()
        thread = th
        val h = Handler(th.looper)
        handler = h
        h.post {
            choreographer = Choreographer.getInstance()
            choreographer?.postFrameCallback(frame)
        }
    }

    private fun stop() {
        running = false
        handler?.post { choreographer?.removeFrameCallback(frame) }
        thread?.quitSafely()
        thread = null
        handler = null
    }

    private fun lock(s: Surface): Canvas? {
        if (hardware) {
            try {
                return s.lockHardwareCanvas()
            } catch (e: Exception) {
                hardware = false
            }
        }
        return try {
            s.lockCanvas(null)
        } catch (e: Exception) {
            null
        }
    }

    private fun draw(dt: Float) {
        val s = surface ?: return
        if (!s.isValid) return
        val c = lock(s) ?: return
        try {
            c.drawColor(Color.rgb(8, 12, 28))
            val st = stable
            val area = if (st.width() > 0 && st.height() > 0) st else Rect(0, 0, width, height)
            val w = area.width().toFloat()
            val h = area.height().toFloat()

            // нижняя полоса с дополнительными значениями
            val footerH = min(max(h * 0.16f, 56f), h * 0.3f)
            val gh = h - footerH

            // сглаживание показаний: постоянная времени 0,12 с
            val k = 1f - exp(-dt / 0.12f)
            for (g in gauges) {
                val raw = NavData.get(g.id)
                g.has = raw != null
                val target = if (raw != null) (raw * g.scale).toFloat() else g.min
                g.shown += (target - g.shown) * k
            }

            // раскладка: ряд или колонка, берём ту, где приборы крупнее
            val rowBase = min(w / 6f, gh / 2f) * 0.9f
            val colBase = min(w / 2f, gh / 6f) * 0.9f
            val column = colBase > rowBase
            val base = if (column) colBase else rowBase
            val cell = if (column) gh / 3f else w / 3f
            for (i in gauges.indices) {
                val cx = if (column) area.exactCenterX() else area.left + cell * (i + 0.5f)
                val cy = if (column) area.top + cell * (i + 0.5f) else area.top + gh / 2f
                val r = if (i == 1) base else base * 0.85f
                drawGauge(c, cx, cy, r, gauges[i])
            }

            drawFooter(c, area.left.toFloat(), area.bottom - footerH, w, footerH)

            if (debugOverlay) {
                stroke.strokeWidth = 2f
                stroke.color = Color.CYAN
                c.drawRect(visible, stroke)
                stroke.color = Color.YELLOW
                c.drawRect(stable, stroke)
                text.color = Color.GRAY
                text.textSize = 22f
                text.textAlign = Paint.Align.LEFT
                c.drawText(debug, area.left + 12f, area.top + 26f, text)
                text.textAlign = Paint.Align.CENTER
            }
        } finally {
            try {
                s.unlockCanvasAndPost(c)
            } catch (e: Exception) {
            }
        }
    }

    private fun shownText(g: Gauge): String {
        if (!g.has) return "--"
        val tenths = g.max <= 10f
        val key = if (tenths) (g.shown * 10f).roundToInt() else g.shown.roundToInt()
        if (key != g.textKey) {
            g.textKey = key
            g.text = if (tenths) "${key / 10},${key % 10}" else key.toString()
        }
        return g.text
    }

    private fun format(v: Double?, decimals: Int): String {
        if (v == null) return "--"
        if (decimals == 0) return v.roundToInt().toString()
        val k = (v * 100.0).roundToInt()
        return "${k / 100},${(k % 100).toString().padStart(2, '0')}"
    }

    private fun drawFooter(c: Canvas, left: Float, top: Float, w: Float, h: Float) {
        for (j in infos.indices) {
            val inf = infos[j]
            val cx = left + w * (j + 0.5f) / infos.size
            text.color = Color.rgb(170, 185, 220)
            text.textSize = h * 0.26f
            c.drawText(inf.label, cx, top + h * 0.40f, text)
            val v = NavData.get(inf.id)
            text.color = Color.WHITE
            text.textSize = h * 0.40f
            c.drawText(if (v == null) "--" else format(v, inf.decimals) + " " + inf.unit, cx, top + h * 0.86f, text)
        }
    }

    private fun drawGauge(c: Canvas, cx: Float, cy: Float, r: Float, g: Gauge) {
        val frac = ((g.shown - g.min) / (g.max - g.min)).coerceIn(0f, 1f)

        fill.color = Color.rgb(18, 26, 54)
        c.drawCircle(cx, cy, r, fill)

        val ring = r * 0.86f
        rectF.set(cx - ring, cy - ring, cx + ring, cy + ring)
        stroke.strokeWidth = r * 0.07f
        stroke.strokeCap = Paint.Cap.ROUND
        stroke.color = Color.rgb(40, 52, 90)
        c.drawArc(rectF, 135f, 270f, false, stroke)
        stroke.color = if (frac > 0.85f) Color.rgb(255, 80, 60) else Color.rgb(60, 170, 255)
        c.drawArc(rectF, 135f, 270f * frac, false, stroke)
        stroke.strokeCap = Paint.Cap.BUTT

        // деления
        stroke.strokeWidth = r * 0.02f
        stroke.color = Color.rgb(170, 185, 220)
        val ticks = 10
        for (i in 0..ticks) {
            val a = (135.0 + 270.0 * i / ticks) * PI / 180.0
            val ca = cos(a).toFloat()
            val sa = sin(a).toFloat()
            c.drawLine(cx + r * 0.70f * ca, cy + r * 0.70f * sa, cx + r * 0.78f * ca, cy + r * 0.78f * sa, stroke)
        }

        // стрелка
        val na = (135.0 + 270.0 * frac) * PI / 180.0
        stroke.strokeWidth = r * 0.035f
        stroke.color = Color.rgb(255, 190, 60)
        c.drawLine(cx, cy, cx + r * 0.66f * cos(na).toFloat(), cy + r * 0.66f * sin(na).toFloat(), stroke)
        fill.color = Color.rgb(255, 190, 60)
        c.drawCircle(cx, cy, r * 0.05f, fill)

        // подписи
        text.color = Color.WHITE
        text.textSize = r * 0.30f
        c.drawText(shownText(g), cx, cy + r * 0.45f, text)
        text.color = Color.rgb(170, 185, 220)
        text.textSize = r * 0.13f
        c.drawText(g.unit, cx, cy + r * 0.62f, text)
        text.textSize = r * 0.14f
        c.drawText(g.label, cx, cy + r * 0.86f, text)
    }

package com.aatorque.obd

import android.Manifest
import android.annotation.SuppressLint
import android.app.Notification
import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.PendingIntent
import android.app.Service
import android.bluetooth.BluetoothAdapter
import android.bluetooth.BluetoothDevice
import android.content.Context
import android.content.Intent
import android.content.pm.PackageManager
import android.content.pm.ServiceInfo
import android.os.Binder
import android.os.Build
import android.os.IBinder
import android.os.IInterface
import android.util.Log
import org.prowl.torque.remote.ITorqueService
import java.lang.reflect.Proxy

/**
 * Локальная замена Torque Pro. Отдаёт тот же интерфейс ITorqueService, но данные берёт
 * напрямую из ELM327. Форк AA Torque привязывается к этому сервису вместо Torque.
 * Работает как foreground-сервис (тип connectedDevice), чтобы система не выгружала его в фоне.
 */
class ObdBridgeService : Service() {
    private var engine: ObdEngine?
        get() = current
        set(v) { current = v }
    @Volatile private var running = false
    @Volatile private var isForeground = false
    private var worker: Thread? = null
    private val prefs by lazy { getSharedPreferences("obd_bridge", MODE_PRIVATE) }

    /**
     * Binder, у которого queryLocalInterface отдаёт наш объект. ITorqueService.Stub.asInterface()
     * в форке вернёт именно его. Прокси реализует интерфейс при любом наборе методов в AIDL:
     * неизвестные методы получают безопасные значения по умолчанию.
     */
    private val binder: IBinder = object : Binder() {
        override fun queryLocalInterface(descriptor: String): IInterface? =
            if (descriptor == DESCRIPTOR) api else null
    }

    @Suppress("UNCHECKED_CAST")
    private val api: ITorqueService by lazy {
        Proxy.newProxyInstance(
            ITorqueService::class.java.classLoader,
            arrayOf(ITorqueService::class.java)
        ) { _, m, args ->
            when (m.name) {
                "asBinder" -> binder
                "listAllPIDs" -> engine?.listIds() ?: MazdaPids.all.map { it.id }.toTypedArray()
                "getPIDInformation" -> {
                    val p = args!![0] as Array<String>
                    Array(p.size) { MazdaPids.byId(p[it])?.info() ?: "" }
                }
                "getPIDValuesAsDouble" -> {
                    val p = args!![0] as Array<String>
                    val e = engine
                    DoubleArray(p.size) { e?.value(p[it]) ?: 0.0 }
                }
                "isConnectedToECU" -> engine != null
                else -> defaultFor(m.returnType)
            }
        } as ITorqueService
    }

    private fun defaultFor(t: Class<*>): Any? = when {
        t == Void.TYPE -> null
        t == Boolean::class.javaPrimitiveType -> false
        t == Int::class.javaPrimitiveType -> 0
        t == Long::class.javaPrimitiveType -> 0L
        t == Float::class.javaPrimitiveType -> 0f
        t == Double::class.javaPrimitiveType -> 0.0
        t.isArray -> java.lang.reflect.Array.newInstance(t.componentType, 0)
        else -> null
    }

    override fun onBind(intent: Intent?): IBinder = binder

    override fun onCreate() {
        super.onCreate()
        running = true
        goForeground(status)    // для сервиса, запущенного только привязкой, может быть запрещено — это нормально
        worker = Thread(::connectLoop, "obd-connect").also { it.isDaemon = true; it.start() }
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        if (intent?.action == ACTION_STOP) {
            stopForeground(STOP_FOREGROUND_REMOVE)
            isForeground = false
            stopSelf()
            return START_NOT_STICKY
        }
        if (!isForeground && !goForeground(status)) {
            // startForegroundService() обязывает вызвать startForeground(); если не вышло — не держим сервис
            stopSelf()
            return START_NOT_STICKY
        }
        return START_STICKY
    }

    override fun onDestroy() {
        running = false
        worker?.interrupt()
        engine?.close(); engine = null
        status = "Остановлено"
        super.onDestroy()
    }

    // ---------------- foreground и уведомление ----------------

    private fun goForeground(text: String): Boolean {
        val n = buildNotification(text)
        return try {
            if (Build.VERSION.SDK_INT >= 29)
                startForeground(NOTIF_ID, n, ServiceInfo.FOREGROUND_SERVICE_TYPE_CONNECTED_DEVICE)
            else startForeground(NOTIF_ID, n)
            isForeground = true
            true
        } catch (e: Exception) {   // Android 12+: старт из фона запрещён; либо нет разрешения/типа в манифесте
            Log.w(TAG, "foreground не запущен: ${e.message}")
            isForeground = false
            false
        }
    }

    @Suppress("DEPRECATION")
    private fun buildNotification(text: String): Notification {
        val nm = getSystemService(NotificationManager::class.java)
        if (nm.getNotificationChannel(CHANNEL) == null) {
            nm.createNotificationChannel(
                NotificationChannel(CHANNEL, "Связь с адаптером OBD", NotificationManager.IMPORTANCE_LOW)
            )
        }
        val open = PendingIntent.getActivity(
            this, 0, Intent(this, ObdSettingsActivity::class.java), PendingIntent.FLAG_IMMUTABLE
        )
        val stop = PendingIntent.getService(
            this, 1, Intent(this, ObdBridgeService::class.java).setAction(ACTION_STOP),
            PendingIntent.FLAG_IMMUTABLE
        )
        return Notification.Builder(this, CHANNEL)
            .setSmallIcon(android.R.drawable.stat_sys_data_bluetooth)
            .setContentTitle("Mazda: диагностика")
            .setContentText(text)
            .setContentIntent(open)
            .addAction(0, "Остановить", stop)
            .setOngoing(true)
            .build()
    }

    /** Обновляет статус для настроек и текст уведомления. */
    private fun post(text: String) {
        status = text
        if (isForeground) {
            getSystemService(NotificationManager::class.java).notify(NOTIF_ID, buildNotification(text))
        }
    }

    // ---------------- подключение ----------------

    /** Канал связи и подсказка на случай, если его не удалось открыть. */
    private class Target(val link: ObdLink, val label: String, val hint: String)

    /** Создаёт канал по выбранному типу подключения; если не получилось, причина остаётся в статусе. */
    @SuppressLint("MissingPermission")
    private fun createTarget(): Target? {
        when (prefs.getString("link_type", "classic")) {
            "wifi" -> {
                val host = prefs.getString("wifi_host", null)?.trim().orEmpty().ifBlank { "192.168.0.10" }
                val port = prefs.getString("wifi_port", null)?.trim()?.toIntOrNull() ?: 35000
                return Target(
                    WifiLink(applicationContext, host, port), "Wi-Fi $host:$port",
                    "Не удалось подключиться по Wi-Fi. Подключите телефон к сети адаптера (WiFi_OBDII, V-LINK…), " +
                        "проверьте адрес и порт, включите зажигание."
                )
            }
            "ble" -> {
                if (Build.VERSION.SDK_INT >= 31 &&
                    checkSelfPermission(Manifest.permission.BLUETOOTH_CONNECT) != PackageManager.PERMISSION_GRANTED) {
                    post("Нет разрешения BLUETOOTH_CONNECT")
                    return null
                }
                val mac = prefs.getString("ble_mac", null)?.takeIf { it.isNotBlank() }
                if (mac == null) {
                    post("Выберите BLE-адаптер: «Найти BLE-адаптеры»")
                    return null
                }
                val dev = BluetoothAdapter.getDefaultAdapter()?.getRemoteDevice(mac) ?: return null
                return Target(
                    BleLink(applicationContext, dev), "BLE ${dev.name ?: mac}",
                    "Не удалось подключиться по BLE. Включите зажигание, подойдите ближе к адаптеру, " +
                        "закройте другие программы с OBD."
                )
            }
            else -> {
                val dev = findDevice()
                if (dev == null) {
                    post(status.takeIf { it.startsWith("Нет разрешения") } ?: "Адаптер не найден: сопрягите его в Bluetooth")
                    return null
                }
                return Target(
                    Elm327Link(dev), dev.name ?: dev.address,
                    "Не удалось открыть канал Bluetooth. Включите зажигание, закройте другие программы " +
                        "с OBD (CarScanner, Torque, старую версию приложения)."
                )
            }
        }
    }

    private fun connectLoop() {
        while (running) {
            var link: ObdLink? = null      // закрываем при любом сбое, иначе адаптер остаётся занятым нами же
            try {
                reconnectFlag = false
                val t = createTarget()
                if (t == null) {
                    Thread.sleep(5000); continue
                }
                post("Подключение: ${t.label}…")
                link = t.link
                try {
                    t.link.open()
                } catch (ex: java.io.IOException) {
                    throw java.io.IOException(t.hint + " " + ex.message)
                }
                post("Инициализация адаптера…")
                val e = ObdEngine(t.link, prefs)
                e.init()
                engine = e
                post("Подключено")
                while (running && e.failures < 5 && !reconnectFlag) Thread.sleep(1000)
                post(if (reconnectFlag) "Переподключение…" else "Связь потеряна")
                reconnectFlag = false
                engine = null
                e.close()
            } catch (ie: InterruptedException) {
                link?.close()
                break
            } catch (ex: Exception) {
                engine?.close(); engine = null
                link?.close()
                post(ex.message ?: "Ошибка подключения")
                try { Thread.sleep(3000) } catch (_: InterruptedException) { break }
            }
        }
    }

    @SuppressLint("MissingPermission")
    private fun findDevice(): BluetoothDevice? {
        if (Build.VERSION.SDK_INT >= 31 &&
            checkSelfPermission(Manifest.permission.BLUETOOTH_CONNECT) != PackageManager.PERMISSION_GRANTED) {
            status = "Нет разрешения BLUETOOTH_CONNECT"
            return null
        }
        val bonded = BluetoothAdapter.getDefaultAdapter()?.bondedDevices ?: return null
        val mac = prefs.getString("obd_mac", null)?.takeIf { it.isNotBlank() } ?: ObdConfig.ADAPTER_MAC.takeIf { it.isNotBlank() }
        if (mac != null) bonded.firstOrNull { it.address.equals(mac, ignoreCase = true) }?.let { return it }
        val re = Regex("obd|elm|v-?link|vgate|konnwei|icar|veepeak|scan|327", RegexOption.IGNORE_CASE)
        return bonded.firstOrNull { re.containsMatchIn(it.name ?: "") }
    }

    companion object {
        private const val TAG = "ObdBridge"
        private const val DESCRIPTOR = "org.prowl.torque.remote.ITorqueService"
        private const val CHANNEL = "obd_bridge"
        private const val NOTIF_ID = 4711
        private const val ACTION_STOP = "com.aatorque.obd.STOP"

        /** Текст состояния для экрана настроек. */
        @Volatile var status: String = "Не запущено"

        @Volatile private var current: ObdEngine? = null
        @Volatile private var reconnectFlag = false

        /** Текущее значение параметра для экрана проверки. Вызывать только не из главного потока. */
        fun peek(id: String): Double? = current?.value(id)

        /** Переподключиться (например, после смены адаптера). */
        fun reconnect(context: Context) {
            reconnectFlag = true
            start(context)
        }

        /** Остановить foreground-режим и сервис. */
        fun stop(context: Context) {
            try {
                context.startService(Intent(context, ObdBridgeService::class.java).setAction(ACTION_STOP))
            } catch (e: Exception) {
                Log.w(TAG, "stop: ${e.message}")
            }
        }

        /**
         * Запускает сервис как foreground. Безопасно вызывать откуда угодно: если система запрещает
         * старт из фона (Android 12+), исключение гасится, и сервис просто работает как привязанный.
         * Гарантированно разрешено, когда приложение на экране (например, из SettingsActivity).
         */
        fun start(context: Context) {
            try {
                context.startForegroundService(Intent(context, ObdBridgeService::class.java))
            } catch (e: Exception) {
                Log.w(TAG, "start: ${e.message}")
            }
        }
    }
}
